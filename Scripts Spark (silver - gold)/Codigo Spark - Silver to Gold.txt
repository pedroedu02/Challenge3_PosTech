

import sys
from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job
from pyspark.sql import functions as F

args = getResolvedOptions(sys.argv, ['JOB_NAME', 'BUCKET_NAME'])

sc = SparkContext.getOrCreate()
glueContext = GlueContext(sc)
spark = glueContext.spark_session
job = Job(glueContext)
job.init(args['JOB_NAME'], args)

BUCKET = args['BUCKET_NAME']
SILVER_PATH = f"s3://{BUCKET}/Silver/state_of_data/"
GOLD_BASE_PATH = f"s3://{BUCKET}/Gold/"

# --- 1. Leitura da Silver ---
df_silver = spark.read.parquet(SILVER_PATH)
print(f"Silver lida: {df_silver.count()} linhas")

# --- 2. Preencher "Nao informado" nas dimensoes categoricas usadas para cruzar/agrupar ---
# Isso evita que combinacoes com valor nulo (ex.: cargo nao respondido, mas salario sim)
# sumam silenciosamente da tabela Gold - viram uma categoria visivel e explicita.
DIMENSOES_CATEGORICAS = [
    "cargo_atual", "senioridade_comparavel", "regiao",
    "nivel_ensino", "genero", "modelo_trabalho",
]

def preencher_nao_informado(df):
    for c in DIMENSOES_CATEGORICAS:
        if c in df.columns:
            df = df.withColumn(
                c, F.when(F.col(c).isNull(), F.lit("Não informado")).otherwise(F.col(c))
            )
    return df

df_base = preencher_nao_informado(df_silver)

# --- 3. gold_mercado: estrutura do mercado (quantidade de profissionais por perfil) ---
gold_mercado = (
    df_base
    .groupBy("ano_pesquisa", "regiao", "senioridade_comparavel", "cargo_atual", "nivel_ensino")
    .agg(F.count("*").alias("total_profissionais"))
)

# --- 4. gold_remuneracao: faixa salarial por cargo/senioridade/regiao ---
# Filtra apenas quem respondeu faixa salarial (nao faz sentido incluir quem nao respondeu
# nessa tabela especifica - filtro proprio da analise, diferente da limpeza geral da Silver)
gold_remuneracao = (
    df_base
    .filter(F.col("faixa_salarial").isNotNull())
    .groupBy("ano_pesquisa", "cargo_atual", "senioridade_comparavel", "regiao", "faixa_salarial")
    .agg(F.count("*").alias("total_profissionais"))
)

# --- 5. gold_tecnologias: linguagem, cloud e BI mais usados ---
# As 3 colunas de origem sao multi-escolha (varias tecnologias numa celula so, separadas
# por virgula). Usamos explode() para transformar 1 celula "Python, SQL" em 2 linhas
# separadas - senao "Python, SQL" viraria uma categoria diferente de "Python" sozinho.
def explode_tecnologia(df, coluna, categoria):
    return (
        df.filter(F.col(coluna).isNotNull())
          .withColumn("tecnologia", F.explode(F.split(F.col(coluna), r",\s*")))
          .withColumn("categoria", F.lit(categoria))
          .select("ano_pesquisa", "categoria", "tecnologia", "cargo_atual", "senioridade_comparavel")
    )

gold_tecnologias = (
    explode_tecnologia(df_base, "linguagem_preferida", "linguagem")
    .unionByName(explode_tecnologia(df_base, "cloud_dia_a_dia", "cloud"))
    .unionByName(explode_tecnologia(df_base, "ferramenta_bi_dia_a_dia", "bi"))
    .groupBy("ano_pesquisa", "categoria", "tecnologia", "cargo_atual", "senioridade_comparavel")
    .agg(F.count("*").alias("total_mencoes"))
)

# --- 6. gold_ia: adoção e uso de IA generativa ---
gold_ia = (
    df_base
    .filter(F.col("uso_chatgpt_copilot").isNotNull())
    .groupBy("ano_pesquisa", "cargo_atual", "senioridade_comparavel", "uso_chatgpt_copilot")
    .agg(F.count("*").alias("total_profissionais"))
)

# --- 7. gold_diversidade: perfil de genero por senioridade/cargo/regiao ---
gold_diversidade = (
    df_base
    .filter(F.col("genero").isNotNull())
    .groupBy("ano_pesquisa", "genero", "senioridade_comparavel", "cargo_atual", "regiao")
    .agg(F.count("*").alias("total_profissionais"))
)

# --- 8. gold_trabalho: modelo de trabalho x remuneracao x senioridade ---
gold_trabalho = (
    df_base
    .filter(F.col("modelo_trabalho").isNotNull())
    .groupBy("ano_pesquisa", "modelo_trabalho", "faixa_salarial", "senioridade_comparavel")
    .agg(F.count("*").alias("total_profissionais"))
)

# --- 9. Gravacao de todas as tabelas Gold, particionadas por ano_pesquisa ---
TABELAS_GOLD = {
    "mercado": gold_mercado,
    "remuneracao": gold_remuneracao,
    "tecnologias": gold_tecnologias,
    "ia": gold_ia,
    "diversidade": gold_diversidade,
    "trabalho": gold_trabalho,
}

for nome, df in TABELAS_GOLD.items():
    caminho = f"{GOLD_BASE_PATH}{nome}/"
    df.write.mode("overwrite").partitionBy("ano_pesquisa").parquet(caminho)
    print(f"gold_{nome} gravada em {caminho} ({df.count()} linhas)")

job.commit()
